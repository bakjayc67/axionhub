local function isAlive(char)
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    return hum and hum.Health > 0
end
local function getChar(plr) return plr and plr.Character end
local function getRoot(char)
    return char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso"))
end
local function getHead(char) return char and char:FindFirstChild("Head") end
local function sameTeam(plr)
    if LP.Team and plr.Team then return LP.Team == plr.Team end
    return false
end
local function isVisible(part)
    if not part then return false end
    local origin = Camera.CFrame.Position
    local dir = part.Position - origin
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = { LP.Character, Camera }
    params.FilterType = Enum.RaycastFilterType.Exclude
    local result = workspace:Raycast(origin, dir, params)
    return result == nil or result.Instance:IsDescendantOf(part.Parent)
end
local function worldToScreen(pos)
    local v, on = Camera:WorldToViewportPoint(pos)
    return Vector2.new(v.X, v.Y), on, v.Z
end

local function getClosest()
    local closest, closestDist = nil, State.FOVRadius
    local mousePos = UserInputService:GetMouseLocation()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LP then continue end
        if State.TeamCheck and sameTeam(plr) then continue end
        local char = getChar(plr)
        if not isAlive(char) then continue end
        local part = char:FindFirstChild(State.AimPart) or getHead(char) or getRoot(char)
        if not part then continue end
        local screen, onScreen = worldToScreen(part.Position)
        if not onScreen then continue end
        local dist = (screen - mousePos).Magnitude
        if dist >= closestDist then continue end
        if State.VisibleCheck and not isVisible(part) then continue end
        closest = part
        closestDist = dist
    end
    return closest
end

RunService.RenderStepped:Connect(function()
    if not State.EnableAimbot then return end
    local ok = true
    if State.AimCondition == "MouseButton2" then
        ok = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
    elseif State.AimCondition == "MouseButton1" then
        ok = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1)
    end
    if not ok then return end
    local target = getClosest()
    if not target then return end
    if State.AimMethod == "Camera" then
        local goal = CFrame.new(Camera.CFrame.Position, target.Position)
        Camera.CFrame = Camera.CFrame:Lerp(goal, 1 - math.clamp(State.Smoothing, 0, 0.99))
    elseif typeof(mousemoverel) == "function" then
        local screen = worldToScreen(target.Position)
        local mouse = UserInputService:GetMouseLocation()
        local delta = (screen - mouse) * (1 - State.Smoothing)
        mousemoverel(delta.X, delta.Y)
    end
end)

local lastTrigger = 0

    if not State.EnableTriggerbot then return end
    if tick() - lastTrigger < State.TriggerDelay then return end
    local target = getClosest()
    if not target then return end

        local p = Players:GetPlayerFromCharacter(target.Parent)
        if p and sameTeam(p) then return end
    end
    local screen, on = worldToScreen(target.Position)
    local mouse = UserInputService:GetMouseLocation()
    if on and (screen - mouse).Magnitude < 14 then
        if typeof(mouse1click) == "function" then mouse1click() end
        lastTrigger = tick()
    end
end)

local ESPStore = {}
local function clearPlayerESP(plr)
    local store = ESPStore[plr]
    if not store then return end
    for _, v in pairs(store) do
        if typeof(v) == "Instance" then pcall(function() v:Destroy() end)
        elseif type(v) == "table" then
            for _, d in pairs(v) do if d and d.Remove then pcall(function() d:Remove() end) end end
        elseif v and v.Remove then pcall(function() v:Remove() end) end
    end
    ESPStore[plr] = nil
end
local function draw(class, props)
    if not Drawing then return nil end
    local d = Drawing.new(class)
    for k, v in pairs(props or {}) do d[k] = v end
    return d
end

local function updateESP()
    if not State.EnableESP then
        for p in pairs(ESPStore) do clearPlayerESP(p) end
        return
    end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LP then continue end
        if State.ESPTeamCheck and sameTeam(plr) then clearPlayerESP(plr) continue end
        local char = getChar(plr)
        if not isAlive(char) then clearPlayerESP(plr) continue end
        local root = getRoot(char)
        local head = getHead(char)
        if not root or not head then clearPlayerESP(plr) continue end
        local screenHead, onH, zH = worldToScreen(head.Position + Vector3.new(0, 0.4, 0))
        local screenRoot, onR = worldToScreen(root.Position)
        if not onH and not onR then clearPlayerESP(plr) continue end

        local visible = isVisible(root)
        local boxCol = (State.BoxWallCheck and not visible) and State.BoxHiddenColor or State.BoxVisibleColor
        local skelCol = (State.BoxWallCheck and not visible) and State.SkeletonHidden or State.SkeletonVisible

        if not ESPStore[plr] then
            ESPStore[plr] = {
                box = draw("Square", { Thickness = State.Thickness, Filled = false, Visible = true }),
                name = draw("Text", { Size = 13, Center = true, Outline = true, Visible = true }),
                dist = draw("Text", { Size = 12, Center = true, Outline = true, Visible = true }),
                tool = draw("Text", { Size = 12, Center = true, Outline = true, Visible = true }),
                hpBG = draw("Square", { Filled = true, Color = Color3.new(0,0,0), Visible = true }),
                hpFG = draw("Square", { Filled = true, Color = Color3.fromRGB(0,255,80), Visible = true }),
                skeleton = {},
                hl = nil,
            }
            for i = 1, 12 do
                ESPStore[plr].skeleton[i] = draw("Line", { Thickness = State.SkeletonThickness, Visible = true })
            end
        end

        local s = ESPStore[plr]
        local scale = math.clamp(1000 / math.max(zH, 1), 0.35, 8)
        local h = 52 * scale
        local w = 28 * scale
        local boxPos = Vector2.new(screenHead.X - w/2, screenHead.Y)
        local boxSize = Vector2.new(w, h)

        if s.box then
            s.box.Visible = State.BoxESP
            s.box.Position = boxPos
            s.box.Size = boxSize
            s.box.Color = boxCol
            s.box.Thickness = State.Thickness
            s.box.Filled = State.FilledBoxes
            if State.FilledBoxes then s.box.Color = State.FillColor s.box.Transparency = 0.65
            else s.box.Transparency = 0 end
        end
        if s.name then
            s.name.Visible = State.Names
            s.name.Text = plr.Name
            s.name.Position = Vector2.new(screenHead.X, screenHead.Y - 16)
            s.name.Color = State.TextColor
        end
        if s.dist then
            local d = math.floor((root.Position - Camera.CFrame.Position).Magnitude)
            s.dist.Visible = State.Distances
            s.dist.Text = d .. "m"
            s.dist.Position = Vector2.new(screenHead.X, screenHead.Y + h + 2)
            s.dist.Color = State.TextColor
        end
        if s.tool then
            local tool = char:FindFirstChildOfClass("Tool")
            s.tool.Visible = State.Tool and tool ~= nil
            s.tool.Text = tool and tool.Name or ""
            s.tool.Position = Vector2.new(screenHead.X, screenHead.Y + h + 14)
            s.tool.Color = Color3.fromRGB(255, 200, 80)
        end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if s.hpBG and s.hpFG and hum then
            local pct = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
            s.hpBG.Visible = State.HealthBar
            s.hpFG.Visible = State.HealthBar
            s.hpBG.Position = Vector2.new(boxPos.X - 5, boxPos.Y)
            s.hpBG.Size = Vector2.new(3, h)
            s.hpFG.Position = Vector2.new(boxPos.X - 5, boxPos.Y + h * (1 - pct))
            s.hpFG.Size = Vector2.new(3, h * pct)
            s.hpFG.Color = Color3.fromRGB(255 * (1 - pct), 255 * pct, 40)
        end
        if State.Skeleton and s.skeleton then
            local function bone(a, b, idx)
                local line = s.skeleton[idx]
                if not line then return end
                local pa = char:FindFirstChild(a, true)
                local pb = char:FindFirstChild(b, true)
                if pa and pb and pa:IsA("BasePart") and pb:IsA("BasePart") then
                    local sa, oa = worldToScreen(pa.Position)
                    local sb, ob = worldToScreen(pb.Position)
                    line.Visible = oa and ob
                    line.From = sa
                    line.To = sb
                    line.Color = skelCol
                    line.Thickness = State.SkeletonThickness
                else line.Visible = false end
            end
            bone("Head", "UpperTorso", 1)
            bone("UpperTorso", "LowerTorso", 2)
            bone("UpperTorso", "LeftUpperArm", 3)
            bone("LeftUpperArm", "LeftLowerArm", 4)
            bone("LeftLowerArm", "LeftHand", 5)
            bone("UpperTorso", "RightUpperArm", 6)
            bone("RightUpperArm", "RightLowerArm", 7)
            bone("RightLowerArm", "RightHand", 8)
            bone("LowerTorso", "LeftUpperLeg", 9)
            bone("LeftUpperLeg", "LeftLowerLeg", 10)
            bone("LowerTorso", "RightUpperLeg", 11)
            bone("RightUpperLeg", "RightLowerLeg", 12)
            bone("Head", "Torso", 1)
            bone("Torso", "Left Arm", 3)
            bone("Torso", "Right Arm", 6)
            bone("Torso", "Left Leg", 9)
            bone("Torso", "Right Leg", 11)
        elseif s.skeleton then
            for _, line in pairs(s.skeleton) do if line then line.Visible = false end end
        end
        if State.EnableChams and not (State.ChamsTeamCheck and sameTeam(plr)) then
            if not s.hl or not s.hl.Parent then
                local hl = Instance.new("Highlight")
                hl.Name = "CKChams"
                hl.FillColor = State.ChamsColor
                hl.OutlineColor = State.OutlineColor
                hl.FillTransparency = 0.45
                hl.OutlineTransparency = State.GlowOutline and 0 or 1
                hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                hl.Adornee = char
                hl.Parent = char
                s.hl = hl
            else
                s.hl.FillColor = State.ChamsColor
                s.hl.OutlineColor = State.OutlineColor
                s.hl.OutlineTransparency = State.GlowOutline and 0 or 1
                s.hl.Enabled = true
            end
        elseif s.hl then s.hl.Enabled = false end
    end
end

local warnGui
RunService.Heartbeat:Connect(function()
    if not State.AimingWarning then if warnGui then warnGui.Enabled = false end return end
    local hit = false
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LP or (State.TeamCheck and sameTeam(plr)) then continue end
        local char = getChar(plr)
        if not isAlive(char) then continue end
        local head = getHead(char)
        if not head then continue end
        local look = head.CFrame.LookVector
        local toMe = (Camera.CFrame.Position - head.Position).Unit
        if look:Dot(toMe) > 0.9 and (head.Position - Camera.CFrame.Position).Magnitude < 280 then
            hit = true break
        end
    end
    if not warnGui then
        local sg = Instance.new("ScreenGui")
        sg.Name = "CKAimWarn"
        sg.ResetOnSpawn = false
        sg.Parent = (gethui and gethui()) or CoreGui
        local lbl = Instance.new("TextLabel")
        lbl.Size = UDim2.new(0, 260, 0, 26)
        lbl.Position = UDim2.new(0.5, -130, 0.07, 0)
        lbl.BackgroundTransparency = 1
        lbl.Text = "▲ Enemy aiming at you"
        lbl.TextColor3 = Color3.fromRGB(255, 55, 55)
        lbl.Font = Enum.Font.GothamBold
        lbl.TextSize = 15
        lbl.Parent = sg
        warnGui = sg
    end
    warnGui.Enabled = hit
end)

UserInputService.JumpRequest:Connect(function()
    if not State.BunnyHop then return end
    local char = getChar(LP)
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.FloorMaterial == Enum.Material.Air then return end
    hum.Jump = true
    local root = getRoot(char)
    if root then
        local vel = root.AssemblyLinearVelocity
        local flat = Vector3.new(vel.X, 0, vel.Z)
        if flat.Magnitude > 1 then
            root.AssemblyLinearVelocity = flat.Unit * (flat.Magnitude * State.BhopSpeedBoost) + Vector3.new(0, vel.Y, 0)
        end
    end
end)

    end
    if State.EnableCameraFOV then Camera.FieldOfView = State.CameraFOVValue end
end)
